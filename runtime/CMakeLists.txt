if(NOT APPLE)
    add_library(bbrt STATIC hooks.c plugin.c host_hooks.c capture.c probe.c loader.c nid.c)
    target_include_directories(bbrt PUBLIC include)
endif()
# The executable loader on its own, for the verification harness (tools/harness.py loads it).
add_library(pbloader SHARED loader.c nid.c)
target_include_directories(pbloader PUBLIC include)
# The boot harness: runs the executable's entry on our runtime (runtime/boot.c). System library
# implementations are compiled in directly, so the linker keeps every RT_SYSLIB entry.
add_library(pbdirect_memory STATIC direct_memory.c)
target_include_directories(pbdirect_memory PUBLIC include)
file(GLOB PB_SYSLIB CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/syslib/*.c)
set(PB_BOOT_SUPPORT_SOURCES loader.c memory.c nid.c process.c module.c cpu.c logical_cpu.c host_hooks.c ${PB_SYSLIB})
if(APPLE)
    list(APPEND PB_BOOT_SUPPORT_SOURCES boot_darwin.c guest_tls_darwin.c)
endif()
add_executable(pbboot boot.c ${PB_BOOT_SUPPORT_SOURCES})
# Tests may compile the same boot translation unit directly, without a production injection API.
set(PB_BOOT_SUPPORT_SOURCES_ABSOLUTE "")
foreach(source IN LISTS PB_BOOT_SUPPORT_SOURCES)
    if(IS_ABSOLUTE "${source}")
        list(APPEND PB_BOOT_SUPPORT_SOURCES_ABSOLUTE "${source}")
    else()
        list(APPEND PB_BOOT_SUPPORT_SOURCES_ABSOLUTE "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
    endif()
endforeach()
set(PB_BOOT_SUPPORT_SOURCES ${PB_BOOT_SUPPORT_SOURCES_ABSOLUTE} PARENT_SCOPE)
target_include_directories(pbboot PRIVATE include)
target_link_libraries(pbboot PRIVATE ${CMAKE_DL_LIBS} pbdirect_memory Threads::Threads)
