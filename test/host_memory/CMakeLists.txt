cmake_minimum_required(VERSION 3.20)
project(paleblood_host_memory C)
set(CMAKE_C_STANDARD 11)
add_compile_options(-Wall -Wextra -Werror)
enable_testing()
get_filename_component(PB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(test_host_memory "${PB_ROOT}/test/test_host_memory.c" "${PB_ROOT}/runtime/process.c" "${PB_ROOT}/runtime/syslib/table.c")
target_include_directories(test_host_memory PRIVATE "${PB_ROOT}/runtime/include")
add_test(NAME host_memory COMMAND test_host_memory)
